PassProve Widget - Ukázka použití bez Reactu

Tato stránka demonstruje, jak snadno přidat PassProve widget pro ověření věku do vašeho e-shopu.

Jednoduchá implementace

Přidejte widget do vašeho e-shopu s minimálním množstvím kódu:

<!-- 1. Přidejte skript PassProve do vašeho HTML -->
<script src="https://cdn.passprove.com/passprove-widget.js"></script>

<!-- 2. Přidejte kontejner pro umístění widgetu -->
<div id="passprove-container"></div>

<!-- 3. Inicializujte widget -->
<script>
  document.addEventListener('DOMContentLoaded', function() {
    PassProve.init({
      selector: '#passprove-container',
      shopId: 'vas-eshop-id',  // Povinný parametr
      primaryColor: '#2563eb', // Volitelné přizpůsobení
      buttonText: 'Ověřit věk'
    });
  });
</script>

Přizpůsobení widgetu

Widget můžete snadno přizpůsobit vlastním potřebám:

PassProve.init({
  selector: '#passprove-container-custom',
  shopId: 'vas-eshop-id',
  primaryColor: '#10b981',    // Zelená barva
  secondaryColor: '#d1fae5',  // Světle zelená barva
  buttonShape: 'pill',        // Zaoblené tlačítko (možnosti: 'rounded', 'square', 'pill')
  buttonText: 'Potvrdit věk 18+',
  showBankID: true,           // Můžete vypnout metody ověření, které nechcete používat
  showFaceScan: false,
  onVerificationSuccess: function(method) {
    console.log('Uživatel byl úspěšně ověřen metodou: ' + method);
    // Zde můžete přidat vlastní kód, který se spustí po úspěšném ověření
  }
});
